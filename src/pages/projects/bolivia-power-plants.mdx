import PlantMap from "../../components/viz/bolivia-power-plants/PlantMap.astro";
import GenerationHistory from "../../components/viz/bolivia-power-plants/GenerationHistory.astro";
import Html from "../../components/Html.astro";
import { content } from "../../lib/content";
import d from "../../data/bolivia-power-plants/plants.json";
import h from "../../data/bolivia-power-plants/history.json";

{/* All visible text lives in content/projects/bolivia-power-plants.md. This file computes the numbers and places the text. */}
export const t = d.meta.totals;
export const top3 = d.departments.slice(0, 3);
export const share = (k) => Math.round((top3.reduce((a, x) => a + x[k], 0) / d.departments.reduce((a, x) => a + x[k], 0)) * 100);
export const fmt = (n) => Math.round(n).toLocaleString("en-US");
export const p = Object.fromEntries(d.plants.map((x) => [x.name, x]));
export const h0 = h.series[0];
export const h1 = h.series[h.series.length - 1];
export const peak = h.series.reduce((a, x) => (x.thermal / x.total > a.thermal / a.total ? x : a));
export const pct = (a, b) => Math.round((a / b) * 100);
export const sum = (arr, f) => arr.reduce((a, x) => a + f(x), 0);
export const isGas = (x) => ["gas_cc", "gas_oc", "diesel", "dual_fuel"].includes(x.tech);
export const capBy = (k) => sum(d.plants, (x) => x.capacity_by_tech_mw[k] ?? 0);
export const gasMW = capBy("gas_cc") + capBy("gas_oc") + capBy("diesel") + capBy("dual_fuel");
export const zongo = ["Zongo", "Tiquimani", "Botijlaca", "Cuticucho", "Santa Rosa", "Sainani", "Chururaqui", "Harca", "Cahua", "Huaji"].map((n) => p[n]);
export const zMW = sum(zongo, (x) => x.capacity_mw);
export const zGWh = sum(zongo, (x) => x.generation_2025_gwh);
export const gasCF = sum(d.plants.filter(isGas), (x) => x.generation_2025_gwh) * 1000 / (sum(d.plants.filter(isGas), (x) => x.capacity_mw) * 8760);
export const dep = Object.fromEntries(d.departments.map((x) => [x.department, x]));
export const approx = d.plants.filter((x) => x.location_precision === "approximate");
export const txt = content("projects/bolivia-power-plants", {
  plants: t.plants,
  capacity_mw: fmt(t.capacity_mw),
  generation_gwh: fmt(t.generation_2025_gwh),
  top3_capacity_pct: share("capacity_mw"),
  top3_generation_pct: share("generation_2025_gwh"),
  gas_mw: fmt(gasMW),
  gas_pct: pct(gasMW, t.capacity_mw),
  hydro_mw: fmt(capBy("hydro")),
  gas_cf: Math.round(gasCF * 100),
  gas_cc_mw: fmt(capBy("gas_cc")),
  gas_oc_mw: fmt(capBy("gas_oc")),
  gas_oc_cf: pct(d.meta.generation_2025_gwh_by_type.gas_oc * 1000, capBy("gas_oc") * 8760),
  engines_mw: (capBy("diesel") + capBy("dual_fuel")).toFixed(1),
  carrasco_mw: fmt(p["Carrasco"].capacity_mw),
  carrasco_gwh: p["Carrasco"].generation_2025_gwh.toFixed(1),
  zongo_n: zongo.length,
  zongo_mw: fmt(zMW),
  zongo_cf: pct(zGWh * 1000, zMW * 8760),
  santa_cruz_gwh: fmt(dep["Santa Cruz"].generation_2025_gwh),
  cochabamba_gwh: fmt(dep["Cochabamba"].generation_2025_gwh),
  la_paz_mw: fmt(dep["La Paz"].capacity_mw),
  la_paz_plants: d.plants.filter((x) => x.department === "La Paz").length,
  approx_n: approx.length,
  approx_mw: fmt(sum(approx, (x) => x.capacity_mw)),
  first_year: h0.year,
  first_gwh: fmt(h0.total),
  first_gas_pct: pct(h0.thermal, h0.total),
  peak_year: peak.year,
  peak_gas_pct: pct(peak.thermal, peak.total),
  last_year: h1.year,
  last_gwh: fmt(h1.total),
  last_gas_pct: pct(h1.thermal, h1.total),
  growth: (h1.total / h0.total).toFixed(1),
});
export const notes = txt.columns("notes");
