export const t = d.meta.totals;
export const top3 = d.departments.slice(0, 3);
export const share = (k) => Math.round((top3.reduce((a, x) => a + x[k], 0) / d.departments.reduce((a, x) => a + x[k], 0)) * 100);
export const fmt = (n) => Math.round(n).toLocaleString("en-US");
export const p = Object.fromEntries(d.plants.map((x) => [x.name, x]));
export const h0 = h.series[0];
export const h1 = h.series[h.series.length - 1];
export const peak = h.series.reduce((a, x) => (x.thermal / x.total > a.thermal / a.total ? x : a));
export const pct = (a, b) => Math.round((a / b) * 100);
export const sum = (arr, f) => arr.reduce((a, x) => a + f(x), 0);
export const isGas = (x) => ["gas_cc", "gas_oc", "diesel", "dual_fuel"].includes(x.tech);
export const capBy = (k) => sum(d.plants, (x) => x.capacity_by_tech_mw[k] ?? 0);
export const gasMW = capBy("gas_cc") + capBy("gas_oc") + capBy("diesel") + capBy("dual_fuel");
export const hydroMW = capBy("hydro");
export const gt = d.plants.filter((x) => x.capacity_by_tech_mw.gas_oc);
export const zongo = ["Zongo", "Tiquimani", "Botijlaca", "Cuticucho", "Santa Rosa", "Sainani", "Chururaqui", "Harca", "Cahua", "Huaji"].map((n) => p[n]);
export const zMW = sum(zongo, (x) => x.capacity_mw);
export const zGWh = sum(zongo, (x) => x.generation_2025_gwh);
export const gasCF = sum(d.plants.filter(isGas), (x) => x.generation_2025_gwh) * 1000 / (sum(d.plants.filter(isGas), (x) => x.capacity_mw) * 8760);
export const dep = Object.fromEntries(d.departments.map((x) => [x.department, x]));
export const lpz = d.plants.filter((x) => x.department === "La Paz");
export const approx = d.plants.filter((x) => x.location_precision === "approximate");

## Notes and method

<div class="method-grid">
<div>

### What is counted

Only the national grid (Sistema Interconectado Nacional, SIN), which the CNDC dispatches. Isolated systems and self-producers are left out. **Capacity** is the CNDC figure for the end of 2025: effective capacity, with thermal units rated at the mean temperature of their site. It is not nameplate capacity. **Generation** is gross generation in 2025, before the plants' own consumption. The **capacity factor** is generation divided by capacity times 8,760 hours, using year-end capacity.

</div>
<div>

### Technologies

Natural gas groups combined cycles ({fmt(capBy("gas_cc"))} MW), open-cycle gas turbines ({fmt(capBy("gas_oc"))} MW), and the diesel and dual-fuel engines at Moxos and Aranjuez ({(capBy("diesel") + capBy("dual_fuel")).toFixed(1)} MW). Biomass plants are at sugar mills. A plant that mixes technologies, such as Guaracachi or Warnes, takes the color of its largest part; the reading card gives the split. Santa Rosa BC and AC, and Uyuni phases I and II, are shown as one plant each.

</div>
<div>

### Locations

Coordinates come from the Global Energy Monitor tracker and from OpenStreetMap. {approx.length} small plants ({fmt(sum(approx, (x) => x.capacity_mw))} MW) have no published position; they are placed at the village or town they belong to and drawn with a dashed outline. Each plant's department is found from its coordinates. The result matches the CNDC's 2025 generation by department to within 0.05 GWh.

</div>
<div>

### Where the sources disagree

The Global Energy Monitor tracker lists the Umapalca (85 MW) and Palillada (118 MW) hydro plants in La Paz as operating. The CNDC does not include them in the grid at the end of 2025, and this map follows the CNDC. The CNDC's [online chart](https://www.cndc.bo/historico-generacion/) gives 9,999.8 GWh for 2021 and 10,604.9 GWh for 2022; its annual report and data files give 9,966.2 and 10,599.5, used here. Six of the CNDC's yearly generation files (2006, 2011, 2013, 2016, 2017, 2019) stop before December, so the history comes from the annual report.

</div>
