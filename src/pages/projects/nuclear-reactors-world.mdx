import ReactorMap from "../../components/viz/nuclear-reactors-world/ReactorMap.astro";
import Html from "../../components/Html.astro";
import { content } from "../../lib/content";
import c from "../../data/nuclear-reactors-world/countries.json";
import s from "../../data/nuclear-reactors-world/sites.json";

{/* All visible text lives in content/projects/nuclear-reactors-world.md. This file computes the numbers and places the text. */}
export const by = Object.fromEntries(c.countries.map((x) => [x.country, x]));
export const china = by["China"];
export const building = c.countries.filter((x) => x.construction > 0);
export const txt = content("projects/nuclear-reactors-world", {
  world_operating: c.world.operating,
  world_construction: c.world.construction,
  world_planned: c.world.planned,
  world_proposed: c.world.proposed,
  world_construction_active: c.world.construction - 6,
  china_construction: china.construction,
  china_planned: china.planned,
  china_sites: s.sites.filter((x) => x.country === "China" && x.units.construction > 0).length,
  others_construction: c.world.construction - china.construction,
  others_countries: building.length - 1,
  firsts_construction: ["Egypt", "Turkey", "Bangladesh"].reduce((a, n) => a + by[n].construction, 0),
  big3_planned: china.planned + by["Russia"].planned + by["India"].planned,
  india_construction: by["India"].construction,
  russia_construction: by["Russia"].construction,
  pakistan_construction: by["Pakistan"].construction,
  us_operating: by["United States"].operating,
  france_operating: by["France"].operating,
  japan_operating: by["Japan"].operating,
  sites: s.sites.length,
  approx_sites: s.sites.filter((x) => x.location === "approximate").length,
});
export const notes = txt.columns("notes");

<div class="rw-intro">
  <div class="prose"><Html html={txt.block("intro")} /></div>
  <div class="tblock rw-kf">{txt.pairs("key-figures").map((k) => <div><small set:html={k.label} /><b set:html={k.value} /></div>)}</div>
</div>

<ReactorMap>
  <ol class="rw-ins" slot="insights">{txt.items("insights").map((h) => <li><span set:html={h} /></li>)}</ol>
</ReactorMap>

<div class="prose after-chart"><Html html={txt.block("after-chart")} /></div>

<section class="method">
  <Html html={notes.head} />
  <div class="method-grid">{notes.cols.map((h) => <div set:html={h} />)}</div>
</section>
